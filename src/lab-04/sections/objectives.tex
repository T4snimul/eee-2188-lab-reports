\section{Objectives}
\begin{enumerate}
    \setlength{\itemsep}{0pt}
          \setlength{\parskip}{0pt}
          \setlength{\parsep}{0pt}
    \item To understand the working principle of full-wave rectifier circuits using both center-tapped transformer and bridge configurations
    \item To construct and analyze a full-wave bridge rectifier circuit using silicon diodes
    \item To measure and compare the input AC voltage with the rectified DC output voltage using an oscilloscope
    \item To study the effect of filter capacitors in reducing output ripple and improving voltage stability in full-wave rectifier circuits
\end{enumerate}
