\section{Calculation}
\markright{Calculation}

\subsubsection{For Square Wave}
\begin{equation}
  V_{\text{rms}} = \frac{V_{\text{pp}}}{2} = \frac{11.44}{2} = 5.72\text{ V}
\end{equation}

\begin{equation}
  \text{Ripple Factor} = \frac{\sqrt{(1.98)^2 - (1.5)^2}}{1.5} = \boxed{0.862}
\end{equation}

\begin{equation}
  \text{Efficiency } (\%) = \frac{(1.98)^2}{(5.72)^2} \times 100 = \boxed{11.98\%}
\end{equation}

\subsubsection{For Triangular Wave}
\begin{equation}
  V_{\text{rms}} = \frac{V_{\text{pp}}}{2\sqrt{3}} = \frac{12.72}{2\sqrt{3}} = 3.67\text{ V}
\end{equation}

\begin{equation}
  \text{Ripple Factor} = \frac{\sqrt{(1.08)^2 - (0.79)^2}}{0.79} = \boxed{0.92}
\end{equation}

\begin{equation}
  \text{Efficiency } (\%) = \frac{(1.08)^2}{(3.67)^2} \times 100 = \boxed{8.66\%}
\end{equation}

\subsubsection{For Sinusoidal Wave}
\begin{equation}
  V_{\text{rms}} = \frac{V_{\text{pp}}}{2\sqrt{2}} = \frac{14.20}{2\sqrt{2}} = 5.0205\text{ V}
\end{equation}

\begin{equation}
  \text{Ripple Factor} = \frac{\sqrt{(1.19)^2 - (0.66)^2}}{0.66} = \boxed{1.02}
\end{equation}

\begin{equation}
  \text{Efficiency } (\%) = \frac{(1.19)^2}{(5.0205)^2} \times 100 = \boxed{5.62\%}
\end{equation}
