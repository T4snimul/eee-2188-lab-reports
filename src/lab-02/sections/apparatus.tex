\section{Apparatus Required}
\markright{Apparatus Required}

\begin{enumerate}
    \item P-n junction diode: 1N4007 silicon diode.
    \item Regulated DC power supply: variable DC output.
    \item Resistor: $10\,\text{k}\Omega$.
    \item Ammeter (0-20 mA)
    \item Voltmeter (0-20 V)
    \item Breadboard.
    \item Connecting wires.
\end{enumerate}
